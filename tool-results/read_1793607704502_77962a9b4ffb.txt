   264→export function AuthDialog() {
   265→  const { authOpen, setAuthOpen, setUser } = useApp();
   266→  const [step, setStep] = useState<"phone" | "otp">("phone");
   267→  const [phone, setPhone] = useState("");
   268→  const [name, setName] = useState("");
   269→  const [code, setCode] = useState("");
   270→  const [devCode, setDevCode] = useState<string | null>(null);
   271→  const [loading, setLoading] = useState(false);
   272→  const [error, setError] = useState("");
   273→
   274→  const close = () => {
   275→    setAuthOpen(false);
   276→    setStep("phone"); setCode(""); setDevCode(null); setError("");
   277→  };
   278→
   279→  const requestOtp = async () => {
   280→    setLoading(true); setError("");
   281→    try {
   282→      const res = await apiPost<{ devCode: string }>("/api/auth", { action: "request-otp", phone });
   283→      setDevCode(res.devCode);
   284→      setStep("otp");
   285→    } catch (e) {
   286→      setError((e as Error).message);
   287→    } finally {
   288→      setLoading(false);
   289→    }
   290→  };
   291→
   292→  const verify = async () => {
   293→    setLoading(true); setError("");
   294→    try {
   295→      const res = await apiPost<{ ok: boolean; user: { id: string; name: string; phone: string; role: string; avatar: string } }>(
   296→        "/api/auth", { action: "verify-otp", phone, code, name }
   297→      );
   298→      const me = await apiGet<{ user: unknown; unreadNotifications: number; unreadMessages: number }>("/api/auth");
   299→      setUser(res.user, me.unreadNotifications || 0, me.unreadMessages || 0);
   300→      close();
   301→      toast({ title: "خوش آمدید! 🎉", description: "ورود شما با موفقیت انجام شد." });
   302→      if (res.user.role === "ADMIN") useApp.getState().goAdmin("overview");
   303→      else useApp.getState().setMode("student");
   304→    } catch (e) {
   305→      setError((e as Error).message);
   306→    } finally {
   307→      setLoading(false);
   308→    }
   309→  };
   310→
   311→  return (
   312→    <Dialog open={authOpen} onOpenChange={(o) => !o && close()}>
   313→      <DialogContent className="max-w-md border-border bg-card sm:rounded-2xl">
   314→        <DialogHeader>
   315→          <DialogTitle className="text-lg font-extrabold">
   316→            {step === "phone" ? "ورود | ثبت‌نام" : "کد تایید را وارد کنید"}
   317→          </DialogTitle>
   318→          <DialogDescription>
   319→            {step === "phone"
   320→              ? "با شماره موبایل وارد شوید یا حساب جدید بسازید."
   321→              : `کد ۶ رقمی ارسال‌شده به ${toFa(phone)} را وارد کنید.`}
   322→          </DialogDescription>
   323→        </DialogHeader>
   324→
   325→        {step === "phone" ? (
   326→          <div className="space-y-4">
   327→            <div className="space-y-2">
   328→              <Label htmlFor="auth-name">نام و نام خانوادگی (اختیاری)</Label>
   329→              <Input id="auth-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="مثلاً علی کریمی" />
   330→            </div>
   331→            <div className="space-y-2">
   332→              <Label htmlFor="auth-phone">شماره موبایل</Label>
   333→              <Input
   334→                id="auth-phone" dir="ltr" inputMode="numeric" className="text-left"
   335→                value={phone} onChange={(e) => setPhone(e.target.value.replace(/[^0-9]/g, ""))}
   336→                placeholder="09xxxxxxxxx" maxLength={11}
   337→              />
   338→            </div>
   339→            {error && <p className="text-sm text-destructive">{error}</p>}
   340→            <Button className="w-full" onClick={requestOtp} disabled={loading || phone.length !== 11}>
   341→              {loading ? "در حال ارسال..." : "دریافت کد تایید"}
   342→            </Button>
   343→            <div className="rounded-xl border border-primary/25 bg-primary/5 p-3 text-xs leading-6 text-muted-foreground">
   344→              <p className="font-medium text-primary">حساب‌های نمایشی:</p>
   345→              <p>دانشجو: <span dir="ltr">09123333333</span> | مدیر: <span dir="ltr">09120000000</span> | منتور: <span dir="ltr">09121111111</span></p>
   346→            </div>
   347→          </div>
   348→        ) : (
   349→          <div className="space-y-4">
   350→            {devCode && (
   351→              <div className="rounded-xl border border-amber-500/30 bg-amber-500/10 p-3 text-center text-sm">
   352→                <p className="text-xs text-amber-400">حالت نمایشی — کد تایید شما:</p>
   353→                <p dir="ltr" className="mt-1 text-2xl font-extrabold tracking-[0.4em] text-amber-300">{toFa(devCode)}</p>
   354→              </div>
   355→            )}
   356→            <div dir="ltr" className="flex justify-center">
   357→              <InputOTP maxLength={6} value={code} onChange={setCode}>
   358→                <InputOTPGroup>
   359→                  <InputOTPSlot index={0} /><InputOTPSlot index={1} /><InputOTPSlot index={2} />
   360→                </InputOTPGroup>
   361→                <InputOTPSeparator />
   362→                <InputOTPGroup>
   363→                  <InputOTPSlot index={3} /><InputOTPSlot index={4} /><InputOTPSlot index={5} />
   364→                </InputOTPGroup>
   365→              </InputOTP>
   366→            </div>
   367→            {error && <p className="text-center text-sm text-destructive">{error}</p>}
   368→            <Button className="w-full" onClick={verify} disabled={loading || code.length !== 6}>
   369→              {loading ? "در حال بررسی..." : "تایید و ورود"}
   370→            </Button>
   371→            <Button variant="ghost" className="w-full" onClick={() => setStep("phone")}>
   372→              تغییر شماره

   374→          </div>

   755→
   756→  const add = (e: React.MouseEvent) => {
   757→    e.stopPropagation();
   758→    if (!user) { setAuthOpen(true); return; }

   765→    <Card className="card-hover group cursor-pointer overflow-hidden border-border bg-card" onClick={() => goPublic("course", course.slug)}>
