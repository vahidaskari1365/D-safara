    32→export function CoursesPage() {

    34→    "/api/public?section=catalog"

    45→      return true;

    50→    <main className="mx-auto max-w-7xl px-4 py-10 md:px-6">

    52→
    53→      <div className="mb-8 flex flex-col gap-3 md:flex-row md:items-center">
    54→        <Input placeholder="جستجوی دوره..." value={q} onChange={(e) => setQ(e.target.value)} className="md:max-w-xs" />
    55→        <Select value={cat} onValueChange={setCat}>
    56→          <SelectTrigger className="md:w-48"><SelectValue placeholder="همه دسته‌ها" /></SelectTrigger>
    57→          <SelectContent>
    58→            <SelectItem value="all">همه دسته‌ها</SelectItem>
    59→            {data?.categories.map((c) => <SelectItem key={c.id} value={c.slug}>{c.title}</SelectItem>)}
    60→          </SelectContent>
    61→        </Select>
    62→        <Select value={level} onValueChange={setLevel}>
    63→          <SelectTrigger className="md:w-40"><SelectValue placeholder="همه سطوح" /></SelectTrigger>
    64→          <SelectContent>
    65→            <SelectItem value="all">همه سطوح</SelectItem>
    66→            <SelectItem value="BEGINNER">مقدماتی</SelectItem>
    67→            <SelectItem value="INTERMEDIATE">متوسط</SelectItem>
    68→            <SelectItem value="ADVANCED">پیشرفته</SelectItem>
    69→          </SelectContent>
    70→        </Select>
    71→        <span className="text-xs text-muted-foreground md:mr-auto">{toFa(filtered.length)} دوره</span>
    72→      </div>
    73→
    74→      {loading ? <GridSkeleton /> : filtered.length === 0 ? (
    75→        <EmptyState icon={<CircleAlert className="size-6" />} title="دوره‌ای یافت نشد" desc="فیلترها را تغییر دهید یا عبارت دیگری جستجو کنید." />

   812→export function CartPage() {
   813→  const { cart, removeFromCart, user, setAuthOpen } = useApp();
   814→  const { toast } = useToast();
   815→  const total = cart.reduce((s, c) => s + c.price, 0);
   816→  const [coupon, setCoupon] = useState("");
   817→  const [discount, setDiscount] = useState(0);
   818→
   819→  const validateCoupon = async () => {
   820→    if (!coupon.trim()) return;
   821→    try {
   822→      const res = await apiPost<{ percentOff: number; message: string }>("/api/checkout", { action: "validate-coupon", code: coupon });
   823→      setDiscount(Math.round((total * res.percentOff) / 100));
   824→      toast({ title: "اعمال شد ✅", description: res.message });
   825→    } catch (e) {
   826→      setDiscount(0);
   827→      toast({ title: "نامعتبر", description: (e as Error).message, variant: "destructive" });
   828→    }
   829→  };
   830→
   831→  const checkout = async () => {
   832→    if (!user) { setAuthOpen(true); return; }
   833→    if (!cart.length) return;
   834→    try {
   835→      const res = await apiPost<{ order: { id: string; orderNo: string } }>("/api/checkout", {
   836→        action: "create-order",
   837→        items: cart.map((c) => ({ type: c.type, slug: c.slug })),
   838→        couponCode: discount > 0 ? coupon.trim().toUpperCase() : null,
   839→      });
   840→      useApp.getState().goPublic("checkout", res.order.id);
   841→    } catch (e) {
   842→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   843→    }
   844→  };
   845→
   846→  return (
   847→    <main className="mx-auto max-w-5xl px-4 py-10 md:px-6">
   848→      <SectionHead kicker="سبد خرید" title="سبد خرید شما" />
   849→      {!cart.length ? (
   850→        <EmptyState
   851→          icon={<ShoppingCart className="size-6" />}
   852→          title="سبد خرید شما خالی است"
   853→          desc="دوره‌های عالی اینجا منتظر شما هستند!"
   854→          action={<Button className="mt-2" onClick={() => useApp.getState().goPublic("courses")}>مشاهده دوره‌ها</Button>}
   855→        />
   856→      ) : (
   857→        <div className="grid gap-6 lg:grid-cols-3">
   858→          <div className="space-y-4 lg:col-span-2">
   859→            {cart.map((item) => (
   860→              <Card key={item.slug} className="border-border bg-card">
   861→                <CardContent className="flex items-center justify-between gap-4 p-5">
   862→                  <div>
   863→                    <p className="font-semibold">{item.title}</p>
   864→                    <p className="text-xs text-muted-foreground">{item.type === "course" ? "دوره آموزشی" : "محصول دیجیتال"}</p>
   865→                  </div>
   866→                  <div className="flex items-center gap-4">
   867→                    <p className={cn("font-bold", item.price === 0 && "text-primary")}>{item.price === 0 ? "رایگان" : faToman(item.price)}</p>
   868→                    <Button variant="ghost" size="icon" aria-label="حذف" onClick={() => { removeFromCart(item.slug); setDiscount(0); }}>
   869→                      <XCircle className="size-5 text-destructive" />
   870→                    </Button>
   871→                  </div>
   872→                </CardContent>
   873→              </Card>
   874→            ))}
   875→          </div>
   876→          <Card className="h-fit border-border bg-card">
   877→            <CardContent className="space-y-4 p-6">
   878→              <p className="font-bold">خلاصه سفارش</p>
   879→              <div className="flex gap-2">
   880→                <Input placeholder="کد تخفیف (مثلاً WELCOME15)" value={coupon} onChange={(e) => setCoupon(e.target.value)} />
   881→                <Button variant="outline" onClick={validateCoupon}>اعمال</Button>
   882→              </div>
   883→              <div className="space-y-2 border-t border-border pt-4 text-sm">
   884→                <div className="flex justify-between"><span className="text-muted-foreground">جمع اقلام</span><span>{faToman(total)}</span></div>
   885→                {discount > 0 && <div className="flex justify-between text-primary"><span>تخفیف</span><span>−{faToman(discount)}</span></div>}
   886→                <div className="flex justify-between border-t border-border pt-2 text-base font-bold">
   887→                  <span>مبلغ قابل پرداخت</span><span className="text-primary">{faToman(total - discount)}</span>
   888→                </div>
   889→              </div>
   890→              <Button className="w-full glow-accent" size="lg" onClick={checkout}>ادامه و پرداخت</Button>
   891→              <p className="flex items-center justify-center gap-1.5 text-xs text-muted-foreground"><ShieldCheck className="size-4 text-primary" /> پرداخت امن از طریق درگاه بانکی</p>
   892→            </CardContent>
   893→          </Card>
   894→        </div>
   895→      )}
   896→    </main>
   897→  );

   900→export function CheckoutPage({ orderId }: { orderId: string }) {

   907→    user ? "/api/student?view=orders" : null
   908→  );
